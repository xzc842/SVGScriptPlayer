config {
  size = 960x640,
  background = "#0b1120",
  defaultEasing = "easeOut"
}

# ---------- 宏：带外圈的圆点 ----------
def dot(color, r) {
  circle {
    at = [0, 0],
    r = @r,
    fill = @color,
    stroke = "#ffffff",
    stroke-width = 3
  }
  circle {
    at = [0, 0],
    r = @r + 8,
    fill = "none",
    stroke = @color,
    stroke-width = 2,
    opacity = 0.5
  }
}

# ---------- 宏：带标签的柱子 ----------
def bar(x, height, color, label) {
  rect {
    at = [@x, 500],
    w = 50,
    h = 0,
    rx = 8,
    ry = 8,
    fill = @color,
    name = "bar_" + @label
  }
  move {
    target = "bar_" + @label,
    to = [@x, 500 - @height],
    duration = 800,
    easing = "easeOutBack"
  }
  rect {
    at = [@x, 500 - @height],
    w = 50,
    h = @height,
    rx = 8,
    ry = 8,
    fill = @color,
    name = "bar2_" + @label
  }
  text {
    at = [@x + 25, 540],
    content = @label,
    size = 14,
    color = "#94a3b8",
    anchor = "middle"
  }
  text {
    at = [@x + 25, 480 - @height],
    content = @height,
    size = 16,
    color = @color,
    anchor = "middle",
    weight = 600
  }
}

# 场景 3：三个圆并列（group 嵌套 group）
# ============================================
tag { name = "circles" }

# 先让柱子淡出
fade { target = "bar2_A", to = 0, duration = 400 }
fade { target = "bar2_B", to = 0, duration = 400 }
fade { target = "bar2_C", to = 0, duration = 400 }
fade { target = "bar2_D", to = 0, duration = 400 }
fade { target = "chartTitle", to = 0, duration = 400 }

# 外层 group：整体居中
group {
  at = [480, 340],
  children = [
    circle {
      at = [0, 0],
      r = 200,
      fill = "none",
      stroke = "#1e293b",
      stroke-width = 2,
      opacity = 0,
      name = "ring"
    },
    fade { target = "ring", to = 1, duration = 600 }
  ]
}

wait { ms = 300 }

# 内层 group：三个圆围绕中心
group {
  at = [480, 340],
  children = [
    dot { at = [0, -120],    color = "#4f46e5", r = 30, name = "c1" },
    dot { at = [104, 60],    color = "#ef4444", r = 30, name = "c2" },
    dot { at = [-104, 60],   color = "#10b981", r = 30, name = "c3" }
  ]
}

tag { name = "balls" }

wait { ms = 300 }

# 让三个圆绕中心旋转（用 move 模拟）
move { target = "c1", to = [104, 60],   duration = 600 }
move { target = "c2", to = [-104, 60],  duration = 600 }
move { target = "c3", to = [0, -120],   duration = 600 }

wait { ms = 300 }

move { target = "c1", to = [-104, 60],  duration = 600 }
move { target = "c2", to = [0, -120],   duration = 600 }
move { target = "c3", to = [104, 60],   duration = 600 }
